using HdmAutomation.Engine;

namespace HdmAutomation.Routines;

public sealed class MyReport : Routine
{
    public override string Name => "My Report";
    public override string Description => "What this routine does.";
    public override string StartWindowTitle => "Current Summaries";

    protected override void Define(StepList steps)
    {
        steps.Add("First step", c =>
        {
            var window = c.Ui.Window("Part of a window title");
            window.BringToFront();
        });

        steps.Add("Second step", c =>
        {
            var window = c.Ui.Window("Part of another window title");
            window.RequireText("Text that proves this is the right window");
        });
    }
}
