using HdmAutomation.Engine;

namespace HdmAutomation.Routines;

public sealed class MyReport : ReportJob
{
    public override string Name => "My Report";
    public override string Description => "What this routine does.";

    protected override void Instructions(Script script)
    {

        script.CopyReportToFile("RAD name", "Another report name", "A3", "Workbook.xlsx", "Another sheet", "B2");
    }
}
